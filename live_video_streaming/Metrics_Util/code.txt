import matplotlib.pyplot as plt
import pandas as pd
from datetime import datetime
import re

# Read the log file
try:
    with open('insta_live_No_Start_then_G_logs.txt', 'r') as f:
        content = f.read()
except FileNotFoundError:
    print("Error: 'insta_live_No_Start_then_G_logs.txt' not found.")
    exit()

# Split content into blocks starting with each Timestamp
blocks = re.split(r'Timestamp: ', content)[1:]  # Skip the empty first part

data = []

# Compile regex patterns for efficiency
# Pattern 1: Finds '50th gpu percentile: Xms' and captures X
gpu_pattern = re.compile(r'50th gpu percentile:\s*(\d+)ms')
# Pattern 2: Finds 'Total GPU memory usage: ..., Y.YY MB' and captures Y.YY
mem_pattern = re.compile(r'Total GPU memory usage:.*,\s*([\d\.]+) MB')

for block in blocks:
    lines = block.splitlines()
    if not lines:
        continue
    
    # Parse timestamp
    timestamp_str = lines[0].strip()
    try:
        ts = datetime.strptime(timestamp_str, '%a %b %d %H:%M:%S %Z %Y')
    except ValueError:
        continue  # Skip invalid timestamps
    
    gpu_ms = None
    mem_mb = None

    # Iterate through lines to find our data
    for line in lines:
        # Check for GPU percentile
        if gpu_ms is None:
            gpu_match = gpu_pattern.search(line)
            if gpu_match:
                try:
                    gpu_ms = float(gpu_match.group(1))
                except ValueError:
                    pass
        
        # Check for Memory usage
        if mem_mb is None:
            mem_match = mem_pattern.search(line)
            if mem_match:
                try:
                    mem_mb = float(mem_match.group(1))
                except ValueError:
                    pass

        # If we've found both, no need to scan remaining lines in this block
        if gpu_ms is not None and mem_mb is not None:
            break
            
    # Append data only if both values were found for that timestamp
    if mem_mb is not None and gpu_ms is not None:
        data.append({'time': ts, 'mem': mem_mb, 'gpu': gpu_ms})

# Create DataFrame
df = pd.DataFrame(data)

# Check if DataFrame is empty
if df.empty:
    print("Error: No data was successfully parsed from the log file.")
    print("Please check the parsing logic against the log file format.")
else:
    # Sort by time just in case logs are out of order
    df = df.sort_values('time')
    
    print("Parsing successful. Generating plot...")

    # --- Plotting ---
    fig, axs = plt.subplots(2, 1, figsize=(15, 10), sharex=True)
    
    fig.suptitle('Instagram Application GPU Performance', fontsize=24, y=1.03)

    # Plot Memory utilization (in MB)
    axs[0].plot(df['time'], df['mem'], 'g-', linewidth=2, label='Total GPU Memory')
    axs[0].set_ylabel('Total GPU Memory (MB)', fontsize=16)
    axs[0].tick_params(axis='y', labelsize=12)
    axs[0].grid(True)
    axs[0].legend()

    # Plot GPU (50th percentile frame time in ms as proxy)
    axs[1].plot(df['time'], df['gpu'], 'r-', linewidth=2, label='50th Pctl GPU Time')
    axs[1].set_ylabel('GPU 50th Pctl Time (ms)', fontsize=16)
    axs[1].set_xlabel('Time', fontsize=16)
    axs[1].tick_params(axis='both', labelsize=12)
    axs[1].grid(True)
    axs[1].legend()

    # Rotate x-ticks for readability
    plt.xticks(rotation=30, ha='right')
    
    # Adjust layout
    plt.tight_layout(rect=[0, 0, 1, 0.98])
    plt.show()
